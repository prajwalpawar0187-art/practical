SET-A

1)create database banking_db
mysql> create database banking_db;

2)use database
mysql> use banking_db;

3)create table accounts
mysql> create table accounts(account_id int primary key,account_number varchar(20),customer_name varchar(100),account_type varchar(30),branch varchar(50),city varchar(50),balance decimal(12,2),credit_score int,status varchar(20),opened_date date);

4)display structure
mysql> describe accounts;

5)display all tables in database
mysql> show tables;
