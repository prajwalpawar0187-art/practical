SET-A

1)create database ecommerce_db
mysql> create database ecommerce_db;

2)use that database
mysql> use ecommerce_db;

3)create table products
mysql> create table products(product_id int primary key,product_name varchar(100),category varchar(50),brand varchar(50),price decimal(10,2),quantity int,city varchar(50),status varchar(20));

4)display structure of table
mysql> describe products;

5)display all tables in this database 
mysql> show tables;
