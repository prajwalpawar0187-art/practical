level 1

1)restaurant which having rating greater than 4.5
mysql> select * from restaurants where rating>4.5;

2)restaurant whose average order values is less than 300
mysql> select * from restaurants where avg_order_value<300;

3)restaurants in pune
mysql> select * from restaurants where city='pune';

4)restaurants having more than 20000 orders
mysql> select * from restaurants where orders_count>20000;

5)restaurants whose delivery time is greater than 40 minutes
mysql> select * from restaurants where est_delivery_time>40;

6)restaurants whose rating is between 4.2 and 4.7
mysql> select * from restaurants where rating>=4.2 and rating<=4.7;

7)restaurants who belong to south Indian,italian,biryani cuisine
mysql> select * from restaurants where cuisine in ('italian','biryani cuisine','south indian');

8)restaurant owned by person whose name contains patil
mysql> select * from restaurants where owner_name like '%patil%';

9)restaurants where brand matches the restaurant name
mysql> select * from restaurants where restaurant_name=brand;

10)restaurant having delivery fees less than 30
mysql> select * from restaurants where delivery_fee<30;
