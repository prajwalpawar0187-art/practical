Level-1

1.movies having rating greater than 8
mysql> select * from movies where rating>8;

2.movies whose price is less than 150
mysql> select * from movies where price<150;

3.find all hindi movies
mysql> select * from movies where language='hindi';

4.movies having views more than 1000000
mysql> select * from movies where views>1000000;

5.movies release after 2015
mysql> select * from movies where release_year>2015;

6.movies whose rating between 8 and 8.5
mysql> select * from movies where rating>=8 and rating<=8.5;

7.movies belongs to action, comedy ,or thriller using in
mysql> select * from movies where genre in ('action','thriller','comedy');

8.movies directed by s.s Rajamouli
mysql> select * from movies where director='s. s. rajamouli';

9.movies produce by hombale films
mysql> select * from movies where production_company='hombale films';

10.movies produce by arka media works
mysql> select * from movies where production_company='arka media works';

Level-2

5 most viewed movies
mysql> select * from movies order by views desc limit 5;

2. 5 lowest viewed movies
mysql> select * from movies order by views asc limit 5;

3.movies from highest rating to lowest rating
mysql> select * from movies order by rating desc;

4.distinct languages of movies
mysql> select distinct language from movies;

5.display Movie and Rating using aliases Movie and IMDb_Rating
mysql> select title as movie,rating as imdb_rating from movies;

6.display movie_title, director and production company
mysql> select title,director,production_company from movies;

7.sort movies by production , and then by rating descending
mysql> select * from movies order by production_company,rating desc;

8. 5 highest-rated movies released after 2015
mysql> select * from movies where release_year>2015 order by rating desc limit 5;

9.top 3 most viewed telugu movies
mysql> select * from movies where language='telugu' order by views desc limit 3;

10.top 5 expensive movies
mysql> select * from movies order by price desc limit 5;
